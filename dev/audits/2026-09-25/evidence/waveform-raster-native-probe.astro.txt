<h1>Waveform fractional-scroll coverage</h1><button id="run">Run pixel checks</button><div id="host" style="width:200px"></div><pre id="result">Ready</pre>
<script>
import {renderWaveformVisualizer} from '../../../../../packages/audio/src/view/render/waveform';
import type {AudioPeaks} from '../../../../../packages/audio/src/core';
const host = document.querySelector<HTMLElement>('#host')!;
const button = document.querySelector<HTMLButtonElement>('#run')!;
const result = document.querySelector<HTMLPreElement>('#result')!;
button.addEventListener('click', async () => {
  button.disabled = true;
  try {
    const data = new Int8Array(4000);
    for(let i=0;i<data.length;i+=2){data[i]=-64;data[i+1]=64;}
    const peaks: AudioPeaks = {sampleRate:1000,channels:1,levels:[{samplesPerPeak:10,data}]};
    const cases = [];
    for (const paint of [{name:'opaque',wave:'#999',progress:'#444',alpha:255},{name:'translucent',wave:'rgba(153,153,153,.5)',progress:'rgba(68,68,68,.5)',alpha:128}]) {
    for (const dpr of [1,1.25,2]) {
      const view=renderWaveformVisualizer(host,peaks,{pixelsPerSecond:100,height:40,devicePixelRatio:dpr,waveColor:paint.wave,progressColor:paint.progress,scrollable:false});
      view.redraw(1.5);
      await new Promise<void>(resolve=>requestAnimationFrame(()=>resolve()));
      const canvas = host.querySelector('canvas')!;
      const samples=[];
      for (const offset of [0,0.125,0.25,0.375,0.5,0.75]) {
        view.setOffset!(1+offset/100);
        const pixels=canvas.getContext('2d')!.getImageData(Math.round(20*dpr),Math.round(20*dpr),Math.round(100*dpr),1).data;
        const alpha=Array.from(pixels).filter((_,i)=>i%4===3);
        samples.push({offset,minAlpha:Math.min(...alpha),maxAlpha:Math.max(...alpha)});
      }
      cases.push({paint:paint.name,dpr,samples,pass:samples.every(s=>s.minAlpha===paint.alpha&&s.maxAlpha===paint.alpha)});
      view.dispose();
    }}
    result.textContent=JSON.stringify({browser:navigator.userAgent,cases,pass:cases.every(c=>c.pass)},null,2);
  } catch(error) {result.textContent=String(error);}
  button.disabled=false;
});
</script></body></html>
