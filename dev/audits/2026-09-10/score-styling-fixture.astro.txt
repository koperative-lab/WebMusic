  <script>
    import {defineAllElements} from '@webmusic/score/play/element';
    import {defineAllViewElements} from '@webmusic/score/view/element';
    import {defineAllAnalysisElements} from '@webmusic/score/analyze/element';
    import {defineAllDemoElements, sampleRack} from '@webmusic/score/play/demos';
    import {loadScoreFromUrl} from '@webmusic/score/io';
    import {Effect} from '@webmusic/score/play/headless';
    defineAllElements(); defineAllViewElements(); defineAllAnalysisElements(); defineAllDemoElements();
    const gallery = document.querySelector<HTMLElement>('#gallery')!;
    document.querySelector('#default')!.addEventListener('click', () => gallery.className = '');
    document.querySelector('#transparent')!.addEventListener('click', () => gallery.className = 'dark');
    document.querySelector('#alternate')!.addEventListener('click', () => gallery.className = 'dark alternate');
    document.querySelector('#narrow')!.addEventListener('click', () => gallery.style.width = '320px');
    document.querySelector('#wide')!.addEventListener('click', () => gallery.style.width = '760px');
    document.querySelector('#uniform')!.addEventListener('click', () => gallery.style.setProperty('--wm-harmony-flow-tone', 'var(--wm-accent)'));
    const player = document.querySelector<any>('#player')!;
    for (const [id, type] of [['hold','noteon'], ['release','noteoff']]) {
      document.querySelector('#' + id)!.addEventListener('click', () => {
        for (const midi of [48, 60, 64, 67]) document.querySelector('#live-input')!.dispatchEvent(new CustomEvent('webscore:' + type, {detail: {midi, velocity: 90, startTime: 0}}));
      });
    }
    const synth = document.querySelector<any>('#synth')!;
    synth.sound = [{name:'Gain', min:0, max:1, value:0.5, apply: () => {}}];
    synth.macros = [{label:'Morph', value:0.4, targets:[]}];
    synth.context = new AudioContext();
    synth.bands = [{frequency:120, gain:2, q:0.8}, {frequency:1000, gain:-3, q:1}, {frequency:6000, gain:1, q:1}];
    synth.effects = [Effect.reverb({wet:0.25})];
    document.querySelector<any>('#rack')!.rack = sampleRack();
    loadScoreFromUrl('/xml/chord-progression.musicxml').then(score => {
      for (const el of document.querySelectorAll<any>('#player, #alias, score-view, sheet-view, chord-analysis, key-analysis, roman-analysis, voice-leading-analysis')) el.score = score;
      document.querySelector('#status')!.textContent = 'Ready';
    }).catch(error => { document.querySelector('#status')!.textContent = String(error); });
  </script>
