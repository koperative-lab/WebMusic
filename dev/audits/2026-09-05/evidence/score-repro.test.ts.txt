import {afterEach, describe, expect, it, vi} from 'vitest';
import {SynthPanelAudioGraph} from '/Users/mamingchen/GitHub/WebMusic/packages/score/src/play/headless/synth-panel-audio';
import {ScorePlayerElement} from '/Users/mamingchen/GitHub/WebMusic/packages/score/src/play/element/score-player';

import {ScoreBuilder, Rational, PartId, VoiceId, Pitch, Duration} from '/Users/mamingchen/GitHub/WebMusic/packages/score/src/core';

customElements.define('audit-score-player', ScorePlayerElement);
const flush = async () => {await Promise.resolve(); await Promise.resolve();};
function score() {
 const builder = new ScoreBuilder();
 const part = builder.addPart({id: PartId('part'), name: 'Part', staves:1});
 builder.addNote(part, {id:builder.newNoteId(), pitch:Pitch.parse('C4'), onsetQuarters:Rational.ZERO, duration:Duration.whole(), voice:VoiceId('voice')});
 return builder.build();
}
afterEach(() => document.body.replaceChildren());

describe('score audit reproduction', () => {
 it('reproduces controller rate overwritten during element mount', async () => {
  const s = score(); const player = new ScorePlayer(s);
  const controller = new PlayerController(player, s); controller.setRate(1.5);
  const before = controller.rate;
  const el = document.createElement('audit-score-player') as ScorePlayerElement;
  el.controller = controller;
  document.body.append(el); await flush();
  console.log('controller before/after mount', before, controller.rate);
  expect(before).toBe(1.5); expect(controller.rate).toBe(1);
  el.remove(); controller.destroy(); player.dispose();
 });
 it('reproduces leaked effect node after second build throws', () => {
  const node = () => ({connect: vi.fn(),disconnect:vi.fn()});
  const context = {createGain: () => node()} as unknown as AudioContext;
  const oldDispose=vi.fn(); const leakedDispose=vi.fn();
  const old = {build:()=>({input:node(),output:node(),dispose:oldDispose})};
  const first = {build:()=>({input:node(),output:node(),dispose:leakedDispose})};
  const failure = {build:()=>{throw new Error('custom build failure');}};
  const graph = new SynthPanelAudioGraph();
  const options = {context, bands:[], sections:['effects'] as any, effectValues:[]};
  graph.rebuild({...options, effects:[old] as any});
  expect(()=>graph.rebuild({...options,effects:[first,failure] as any})).toThrow('custom build failure');
  graph.teardown();
  console.log('previous effect disposed/new first effect disposed', oldDispose.mock.calls.length, leakedDispose.mock.calls.length);
  expect(oldDispose).toHaveBeenCalledTimes(1);
  expect(leakedDispose).not.toHaveBeenCalled();
 });
});

import {ScorePlayerScheduler} from '/Users/mamingchen/GitHub/WebMusic/packages/score/src/play/headless/score-player-scheduler';
it('reproduces future JIT attack firing while AudioContext suspended', async () => {
 vi.useFakeTimers();
 const b = new ScoreBuilder(); const part = b.addPart({id:PartId('p'),name:'p',staves:1});
 b.addNote(part,{id:b.newNoteId(),pitch:Pitch.parse('D4'),onsetQuarters:new Rational(1,5),duration:Duration.quarter(),voice:VoiceId('v')});
 const context = {state:'running',currentTime:0,resume:vi.fn()} as unknown as AudioContext;
 const synth = {noteOn:vi.fn(()=>1),noteOff:vi.fn()} as any;
 const noteOn=vi.fn();
 const scheduler = new ScorePlayerScheduler(b.build(),{
  ensureAudio:()=>({context,synth}),getContext:()=>context,getSynth:()=>synth,
  onCursor:vi.fn(),onNoteOn:noteOn,onNoteOff:vi.fn(),onEnd:vi.fn(),
 },{cursorIntervalMs:30,lookaheadSeconds:0.2,schedulerIntervalMs:25,createTickSource:()=>({start:vi.fn(),stop:vi.fn(),dispose:vi.fn()})});
 await scheduler.play();
 expect(synth.noteOn).not.toHaveBeenCalled();
 (context as any).state='suspended';
 await vi.advanceTimersByTimeAsync(150);
 console.log('suspended JIT attack before audio reaches intended onset 0.1s:',synth.noteOn.mock.calls);
 expect(context.currentTime).toBe(0); expect(synth.noteOn).toHaveBeenCalledTimes(1);
 expect(synth.noteOn.mock.calls[0][2]).toBe(0);
 scheduler.dispose(); vi.useRealTimers();
});
